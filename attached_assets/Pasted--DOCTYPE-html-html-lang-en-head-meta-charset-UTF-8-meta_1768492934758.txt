    <script>
        // Get DOM elements
        const calculateBtn = document.getElementById('calculate-btn');
        const resetBtn = document.getElementById('reset-btn');
        const finalAverageElement = document.getElementById('final-average');
        const subjectResultsContainer = document.getElementById('subject-results');
        
        // Subject data structure
        const subjects = [
            {
                id: 'proba',
                name: 'Probability',
                coef: 4,
                inputs: ['proba-td', 'proba-exam'],
                formula: (td, exam) => (td * 0.4) + (exam * 0.6)
            },
            {
                id: 'stat',
                name: 'Statistics',
                coef: 4,
                inputs: ['stat-td', 'stat-exam'],
                formula: (td, exam) => (td * 0.4) + (exam * 0.6)
            },
            {
                id: 'mna',
                name: 'MNA',
                coef: 4,
                inputs: ['mna-td', 'mna-tp', 'mna-exam'],
                formula: (td, tp, exam) => (td * 0.2) + (tp * 0.2) + (exam * 0.6)
            },
            {
                id: 'system',
                name: 'System',
                coef: 2,
                inputs: ['system-td', 'system-exam'],
                formula: (td, exam) => (td * 0.4) + (exam * 0.6)
            },
            {
                id: 'aed',
                name: 'AED',
                coef: 3,
                inputs: ['aed-td', 'aed-exam'],
                formula: (td, exam) => (td * 0.4) + (exam * 0.6)
            },
            {
                id: 'anglais',
                name: 'English',
                coef: 1,
                inputs: ['anglais-exam'],
                formula: (exam) => exam
            }
        ];
        
        // Calculate button event listener
        calculateBtn.addEventListener('click', calculateAverage);
        
        // Reset button event listener
        resetBtn.addEventListener('click', resetAll);
        
        // Calculate the weighted average
        function calculateAverage() {
            let totalWeightedSum = 0;
            let totalCoefficients = 0;
            const subjectAverages = [];
            
            // Calculate average for each subject
            subjects.forEach((subject, index) => {
                // Get input values for this subject
                const inputValues = subject.inputs.map(inputId => {
                    const inputElement = document.getElementById(inputId);
                    const value = parseFloat(inputElement.value);
                    return isNaN(value) ? 0 : Math.min(20, Math.max(0, value));
                });
                
                // Calculate subject average using the formula
                const subjectAverage = subject.formula(...inputValues);
                subjectAverages.push({
                    name: subject.name,
                    average: subjectAverage,
                    coef: subject.coef,
                    passed: subjectAverage >= 10
                });
                
                // Add to weighted sum for final average
                totalWeightedSum += subjectAverage * subject.coef;
                totalCoefficients += subject.coef;
                
                // Update subject result display
                const subjectResultElement = subjectResultsContainer.children[index];
                const gradeElement = subjectResultElement.querySelector('.subject-result-grade');
                const validationElement = subjectResultElement.querySelector('.grade-validation');
                
                gradeElement.textContent = subjectAverage.toFixed(2);
                gradeElement.style.color = subjectAverage >= 10 ? '#28a745' : '#dc3545';
                
                validationElement.textContent = subjectAverage >= 10 ? 'Passed ✓' : 'Failed ✗';
                validationElement.className = `grade-validation ${subjectAverage >= 10 ? 'valid' : 'invalid'}`;
            });
            
            // Calculate final weighted average
            const finalAverage = totalWeightedSum / totalCoefficients;
            finalAverageElement.textContent = finalAverage.toFixed(2);
            
            // Color code the final average
            if (finalAverage >= 10) {
                finalAverageElement.style.color = '#28a745';
            } else if (finalAverage >= 8) {
                finalAverageElement.style.color = '#ffc107';
            } else {
                finalAverageElement.style.color = '#dc3545';
            }
            
            // Update subject card backgrounds based on results
            updateSubjectCardBackgrounds(subjectAverages);
        }
        
        // Update subject card backgrounds based on averages
        function updateSubjectCardBackgrounds(subjectAverages) {
            subjectAverages.forEach((subject, index) => {
                const subjectCard = document.getElementById(`${subjects[index].id}-card`);
                if (subject.passed) {
                    subjectCard.style.backgroundColor = '#f0fff4';
                    subjectCard.style.borderLeft = '5px solid #28a745';
                } else {
                    subjectCard.style.backgroundColor = '#fff5f5';
                    subjectCard.style.borderLeft = '5px solid #dc3545';
                }
            });
        }
        
        // Reset all inputs and results
        function resetAll() {
            // Clear all input fields
            document.querySelectorAll('input').forEach(input => {
                input.value = '';
            });
            
            // Reset final average display
            finalAverageElement.textContent = '--.--';
            finalAverageElement.style.color = '#333';
            
            // Reset subject result displays
            document.querySelectorAll('.subject-result-grade').forEach(grade => {
                grade.textContent = '--.--';
                grade.style.color = '#333';
            });
            
            document.querySelectorAll('.grade-validation').forEach(validation => {
                validation.textContent = '';
            });
            
            // Reset subject card backgrounds
            subjects.forEach((subject, index) => {
                const subjectCard = document.getElementById(`${subject.id}-card`);
                subjectCard.style.backgroundColor = '#f8f9ff';
                subjectCard.style.borderLeft = 'none';
            });
        }
        
        // Add input validation to limit values between 0 and 20
        document.querySelectorAll('input').forEach(input => {
            input.addEventListener('change', function() {
                let value = parseFloat(this.value);
                if (isNaN(value)) {
                    this.value = '';
                } else if (value < 0) {
                    this.value = 0;
                } else if (value > 20) {
                    this.value = 20;
                } else {
                    this.value = Math.round(value * 100) / 100; // Keep 2 decimal places
                }
            });
        });
        
        // Initialize with some example values for demonstration
        function initializeExample() {
            // Set example values
            document.getElementById('proba-td').value = 14;
            document.getElementById('proba-exam').value = 12;
            document.getElementById('stat-td').value = 16;
            document.getElementById('stat-exam').value = 15;
            document.getElementById('mna-td').value = 13;
            document.getElementById('mna-tp').value = 17;
            document.getElementById('mna-exam').value = 11;
            document.getElementById('system-td').value = 10;
            document.getElementById('system-exam').value = 9;
            document.getElementById('aed-td').value = 15;
            document.getElementById('aed-exam').value = 14;
            document.getElementById('anglais-exam').value = 16;
            
            // Calculate the example average
            setTimeout(calculateAverage, 500);
        }
        
        // Initialize with example values when page loads
        window.addEventListener('load', initializeExample);
    </script>
